add_library(astro_camera_models)
target_include_directories(astro_camera_models PUBLIC include)
target_link_libraries(astro_camera_models PUBLIC astro_core)
function(astro_register_camera_module factory)
  set_property(GLOBAL APPEND PROPERTY ASTRO_CAMERA_FACTORIES "${factory}")
endfunction()
# A model is added by its own folder/CMakeLists, without editing a central switch.
file(GLOB model_manifests CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/models/*/CMakeLists.txt")
foreach(manifest IN LISTS model_manifests)
  get_filename_component(model_dir "${manifest}" DIRECTORY)
  get_filename_component(model_name "${model_dir}" NAME)
  add_subdirectory("${model_dir}" "models/${model_name}")
endforeach()
get_property(factories GLOBAL PROPERTY ASTRO_CAMERA_FACTORIES)
if(NOT factories)
  message(FATAL_ERROR "At least one camera module is required")
endif()
foreach(factory IN LISTS factories)
  string(APPEND ASTRO_MODULE_DECLARATIONS "namespace astro::camera_models { const CameraModule& ${factory}(); }\n")
  string(APPEND ASTRO_MODULE_ENTRIES "&${factory}(),")
endforeach()
configure_file(src/Registry.cpp.in "${CMAKE_CURRENT_BINARY_DIR}/Registry.cpp" @ONLY)
target_sources(astro_camera_models PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/Registry.cpp")
