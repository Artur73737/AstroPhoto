file(GLOB ERFA_SOURCES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/external/erfa/src/*.c")
list(FILTER ERFA_SOURCES EXCLUDE REGEX "/(t_erfa_c|t_erfa_c_extra|erfaversion)\\.c$")
add_library(astro_erfa STATIC ${ERFA_SOURCES})
target_include_directories(astro_erfa PUBLIC "${CMAKE_SOURCE_DIR}/external/erfa/src")
add_library(astro_true_north astronomy/src/TrueNorth.cpp)
target_include_directories(astro_true_north PUBLIC astronomy/include)
target_link_libraries(astro_true_north PRIVATE astro_erfa)
if(ASTRO_BUILD_APP)
 add_executable(astro_true_north_cli tools/cli/src/main.cpp)
 target_include_directories(astro_true_north_cli PRIVATE "${CMAKE_SOURCE_DIR}/external/json/include")
 target_link_libraries(astro_true_north_cli PRIVATE astro_true_north astro_stacking)
endif()
if(BUILD_TESTING)
 add_executable(astro_true_north_tests tests/unit/TrueNorthTests.cpp)
 target_link_libraries(astro_true_north_tests PRIVATE astro_true_north astro_erfa GTest::gtest_main)
 gtest_discover_tests(astro_true_north_tests)
endif()
if(BUILD_TESTING)
 add_executable(astro_dss_mosaic tools/dss_mosaic/src/DssMosaic.cpp)
 target_link_libraries(astro_dss_mosaic PRIVATE astro_true_north)
endif()
if(BUILD_TESTING AND ASTRO_BUILD_APP)
 add_executable(astro_north_dss_tests tests/integration/DssTests.cpp)
 target_link_libraries(astro_north_dss_tests PRIVATE astro_true_north astro_erfa astro_stacking GTest::gtest_main)
 gtest_discover_tests(astro_north_dss_tests)
endif()


if(ASTRO_BUILD_APP)
 add_library(astro_polar_solve astrometry/src/PolarSolve.cpp astrometry/catalog/src/PolarCatalog.cpp astrometry/sources/src/PolarSources.cpp astrometry/matching/src/PolarMatch.cpp)
 target_include_directories(astro_polar_solve PUBLIC astrometry/include astrometry/catalog/include astrometry/sources/include astrometry/matching/include)
 target_link_libraries(astro_polar_solve PUBLIC astro_stacking PRIVATE astro_true_north astro_preview)
 target_link_libraries(astro_true_north_cli PRIVATE astro_polar_solve)
endif()
