set(ASI_ROOT "${CMAKE_SOURCE_DIR}/external/zwo/windows/ASI SDK" CACHE PATH "Official ZWO ASI SDK")
if(NOT EXISTS "${ASI_ROOT}/include/ASICamera2.h")
 message(FATAL_ERROR "Download the official ASI SDK; set ASI_ROOT to its Windows SDK directory")
endif()
add_library(astro_asi_sdk SHARED IMPORTED GLOBAL)
set_target_properties(astro_asi_sdk PROPERTIES IMPORTED_IMPLIB "${ASI_ROOT}/lib/x64/ASICamera2.lib" IMPORTED_LOCATION "${ASI_ROOT}/lib/x64/ASICamera2.dll" INTERFACE_INCLUDE_DIRECTORIES "${ASI_ROOT}/include")
add_library(astro_zwo sdk/src/ZwoCamera.cpp)
target_include_directories(astro_zwo PUBLIC sdk/include)
target_link_libraries(astro_zwo PRIVATE astro_asi_sdk PUBLIC astro_fits)
add_executable(astro_zwo_diagnostics diagnostics/src/Enumerate.cpp)
target_link_libraries(astro_zwo_diagnostics PRIVATE astro_zwo)
add_custom_command(TARGET astro_zwo_diagnostics POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${ASI_ROOT}/lib/x64/ASICamera2.dll" "$<TARGET_FILE_DIR:astro_zwo_diagnostics>")
if(BUILD_TESTING)
 add_executable(astro_zwo_tests tests/unit/ZwoTests.cpp)
 target_link_libraries(astro_zwo_tests PRIVATE astro_zwo GTest::gtest_main)
 add_custom_command(TARGET astro_zwo_tests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${ASI_ROOT}/lib/x64/ASICamera2.dll" "$<TARGET_FILE_DIR:astro_zwo_tests>")
 gtest_discover_tests(astro_zwo_tests)
endif()

add_library(astro_zwo_preview preview/src/SensorPreview.cpp)
target_include_directories(astro_zwo_preview PUBLIC preview/include)
target_link_libraries(astro_zwo_preview PUBLIC astro_zwo PRIVATE windowscodecs ole32)
add_library(astro_zwo_adapter adapter/src/ZwoAdapter.cpp)
target_include_directories(astro_zwo_adapter PUBLIC adapter/include)
target_link_libraries(astro_zwo_adapter PUBLIC astro_core PRIVATE astro_zwo astro_zwo_preview)
