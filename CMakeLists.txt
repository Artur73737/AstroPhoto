cmake_minimum_required(VERSION 3.24)
project(AstroCanonController VERSION 0.3.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(ASTRO_BUILD_APP "Build Windows desktop application" ON)
option(ASTRO_WITH_EDSDK "Build the real Canon backend (requires licensed SDK)" OFF)
option(ASTRO_HARDWARE_TESTS "Build opt-in tests using a physical camera" OFF)
include(CTest)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
if(MSVC)
  add_compile_options(/W4 /permissive- /utf-8)
endif()
add_library(astro_core src/core/src/Core.cpp src/core/src/StarAutofocus.cpp)
target_include_directories(astro_core PUBLIC src/core/include)
add_library(astro_configuration src/configuration/src/ConfigManager.cpp)
target_include_directories(astro_configuration PUBLIC src/configuration/include PRIVATE external/json/include)
target_link_libraries(astro_configuration PUBLIC astro_core)
add_library(astro_infrastructure src/infrastructure/src/Infrastructure.cpp)
target_include_directories(astro_infrastructure PUBLIC src/infrastructure/include)
target_link_libraries(astro_infrastructure PUBLIC astro_core astro_configuration)
if(ASTRO_WITH_EDSDK)
  if(NOT WIN32)
    message(FATAL_ERROR "The Canon adapter currently targets Windows")
  endif()
  find_package(EDSDK REQUIRED)
  add_library(astro_edsdk src/edsdk_wrapper/src/EdsCamera.cpp src/edsdk_wrapper/src/PropertyLabels.cpp)
  target_include_directories(astro_edsdk PUBLIC src/edsdk_wrapper/include)
  target_link_libraries(astro_edsdk PUBLIC astro_core PRIVATE EDSDK::EDSDK ole32)
  target_link_libraries(astro_infrastructure PRIVATE astro_edsdk)
  target_compile_definitions(astro_infrastructure PRIVATE ASTRO_WITH_EDSDK)
endif()
if(ASTRO_BUILD_APP)
  if(NOT WIN32)
    message(FATAL_ERROR "Desktop application requires Windows")
  endif()
  set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
  add_subdirectory(external/glfw EXCLUDE_FROM_ALL)
  find_package(OpenGL REQUIRED)
  add_library(imgui external/imgui/imgui.cpp external/imgui/imgui_draw.cpp external/imgui/imgui_tables.cpp external/imgui/imgui_widgets.cpp external/imgui/backends/imgui_impl_glfw.cpp external/imgui/backends/imgui_impl_opengl3.cpp external/imgui/misc/cpp/imgui_stdlib.cpp)
  target_include_directories(imgui PUBLIC external/imgui external/imgui/backends external/imgui/misc/cpp)
  target_link_libraries(imgui PUBLIC glfw OpenGL::GL)
  add_library(astro_ui src/ui/src/MainWindow.cpp src/ui/src/ConnectionPanel.cpp src/ui/src/LiveViewPanel.cpp src/ui/src/FocusPanel.cpp src/ui/src/SequencePanel.cpp src/ui/src/LogPanel.cpp src/ui/src/Translation.cpp)
  target_include_directories(astro_ui PUBLIC src/ui/include)
  target_link_libraries(astro_ui PUBLIC astro_core PRIVATE imgui)
  file(GLOB_RECURSE LIBRAW_SOURCES CONFIGURE_DEPENDS external/libraw/src/*.cpp)
  list(FILTER LIBRAW_SOURCES EXCLUDE REGEX "_ph\\.cpp$")
  add_library(astro_libraw STATIC ${LIBRAW_SOURCES})
  target_include_directories(astro_libraw PUBLIC external/libraw)
  target_compile_definitions(astro_libraw PUBLIC LIBRAW_NODLL PRIVATE LIBRAW_BUILDLIB)
  target_compile_options(astro_libraw PRIVATE /W0)
  add_library(astro_preview src/app/PhotoPreview.cpp src/app/FocusPeaking.cpp src/app/StarPeaking.cpp src/app/StarAnalysis.cpp src/app/LiveViewPreview.cpp src/app/ImageAnalysis.cpp)
  target_include_directories(astro_preview PUBLIC src/app)
  target_link_libraries(astro_preview PUBLIC astro_core PRIVATE astro_libraw windowscodecs ole32)
  enable_language(RC)
  configure_file(src/app/app.rc.in "${CMAKE_CURRENT_BINARY_DIR}/astrophoto.rc" @ONLY)
  add_executable(AstroCanonController WIN32 src/app/main.cpp src/app/app.manifest "${CMAKE_CURRENT_BINARY_DIR}/astrophoto.rc")
  target_link_libraries(AstroCanonController PRIVATE astro_ui astro_infrastructure astro_preview imgui windowscodecs ole32)
  add_custom_command(TARGET AstroCanonController POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/src/configuration/defaults.json" "$<TARGET_FILE_DIR:AstroCanonController>/defaults.json")
  add_custom_command(TARGET AstroCanonController POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/external/imgui/misc/fonts/Cousine-Regular.ttf" "$<TARGET_FILE_DIR:AstroCanonController>/Cousine-Regular.ttf")
  add_custom_command(TARGET AstroCanonController POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:AstroCanonController>/licenses/LibRaw"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/external/libraw/LICENSE.CDDL" "${CMAKE_SOURCE_DIR}/external/libraw/COPYRIGHT" "$<TARGET_FILE_DIR:AstroCanonController>/licenses/LibRaw")
  if(ASTRO_WITH_EDSDK)
    astro_copy_edsdk_runtime(AstroCanonController)
  endif()
endif()
if(BUILD_TESTING)
  set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
  add_subdirectory(external/googletest EXCLUDE_FROM_ALL)
  add_executable(astro_tests tests/unit/CoreTests.cpp tests/unit/ConfigTests.cpp tests/unit/StarAutofocusTests.cpp tests/unit/WorkerTests.cpp)
  target_link_libraries(astro_tests PRIVATE astro_infrastructure GTest::gtest_main)
  target_compile_definitions(astro_tests PRIVATE ASTRO_SOURCE_DIR="${CMAKE_SOURCE_DIR}")
  include(GoogleTest)
  if(ASTRO_WITH_EDSDK)
    astro_copy_edsdk_runtime(astro_tests)
    add_executable(astro_property_tests tests/unit/PropertyLabelsTests.cpp)
    target_link_libraries(astro_property_tests PRIVATE astro_edsdk GTest::gtest_main)
    astro_copy_edsdk_runtime(astro_property_tests)
    gtest_discover_tests(astro_property_tests)
  endif()
  gtest_discover_tests(astro_tests)
  if(ASTRO_BUILD_APP)
    add_executable(astro_ui_tests tests/unit/UiTests.cpp)
    target_link_libraries(astro_ui_tests PRIVATE astro_ui imgui GTest::gtest_main)
    gtest_discover_tests(astro_ui_tests)
    add_executable(astro_preview_tests tests/unit/PreviewTests.cpp tests/unit/FocusPeakingTests.cpp tests/unit/StarAnalysisTests.cpp tests/unit/LiveViewPreviewTests.cpp tests/unit/ImageAnalysisTests.cpp)
    target_link_libraries(astro_preview_tests PRIVATE astro_preview GTest::gtest_main)
    gtest_discover_tests(astro_preview_tests)
  endif()
  if(ASTRO_HARDWARE_TESTS)
    if(NOT ASTRO_WITH_EDSDK)
      message(FATAL_ERROR "Hardware tests require ASTRO_WITH_EDSDK=ON")
    endif()
    add_executable(astro_hardware_tests tests/integration/CameraTests.cpp)
    target_link_libraries(astro_hardware_tests PRIVATE astro_infrastructure GTest::gtest_main)
    if(ASTRO_BUILD_APP)
      target_link_libraries(astro_hardware_tests PRIVATE astro_preview)
      target_compile_definitions(astro_hardware_tests PRIVATE ASTRO_PREVIEW_TESTS)
    endif()
    astro_copy_edsdk_runtime(astro_hardware_tests)
    add_test(NAME hardware COMMAND astro_hardware_tests)
    set_tests_properties(hardware PROPERTIES LABELS hardware SKIP_RETURN_CODE 77)
  endif()
endif()
