cmake_minimum_required(VERSION 3.24)
project(AstroCanonController VERSION 0.10.2 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(ASTRO_BUILD_APP "Build Windows desktop application" ON)
option(ASTRO_WITH_EDSDK "Build the real Canon backend (requires licensed SDK)" OFF)
option(ASTRO_HARDWARE_TESTS "Build opt-in tests using a physical camera" OFF)
set(ASTRO_UPDATE_REPOSITORY "Artur73737/AstroPhoto" CACHE STRING "GitHub owner/repository used for update checks")
include(CTest)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
if(MSVC)
  add_compile_options(/W4 /permissive- /utf-8)
endif()
option(ASTRO_WITH_ZWO "Build the real ZWO ASI backend" ON)
if(BUILD_TESTING)
 set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
 add_subdirectory(external/googletest EXCLUDE_FROM_ALL)
 include(GoogleTest)
endif()
add_subdirectory(src/imaging/fits)
add_subdirectory(src/core)
add_subdirectory(src/localization)
add_subdirectory(src/cameras/models)
add_subdirectory(src/configuration)
if(ASTRO_WITH_EDSDK)
 if(NOT WIN32)
  message(FATAL_ERROR "Canon adapter requires Windows")
 endif()
 find_package(EDSDK REQUIRED)
 add_subdirectory(src/cameras/canon/edsdk)
endif()
add_subdirectory(src/infrastructure)
if(ASTRO_BUILD_APP)
 if(NOT WIN32)
  message(FATAL_ERROR "Desktop application requires Windows")
 endif()
 include(cmake/DesktopDependencies.cmake)
 add_subdirectory(src/stacking)
 add_subdirectory(src/imaging)
 add_subdirectory(src/control)
 add_subdirectory(src/mcp)
 add_subdirectory(src/remote)
 add_subdirectory(src/services/updates)
endif()
if(ASTRO_WITH_ZWO)
 add_subdirectory(src/cameras/zwo)
endif()
add_subdirectory(src/features/true_north)
if(ASTRO_BUILD_APP)
 if(ASTRO_WITH_ZWO)
  add_subdirectory(src/features/true_north/workspace)
 endif()
 add_subdirectory(src/ui)
 add_subdirectory(src/app/command_line)
 add_subdirectory(src/app)
endif()
if(WIN32)
 add_subdirectory(src/installer)
endif()
if(BUILD_TESTING)
 add_subdirectory(tests)
endif()
include(cmake/HeaderChecks.cmake)
